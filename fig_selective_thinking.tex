  stbox/.style={draw=black!70,rounded corners=2pt,fill=black!3,
    align=center,inner sep=4pt,minimum height=34pt},
  stflow/.style={-{Stealth[length=4pt,width=3pt]},draw=black!75,line width=.65pt},
  stcheck/.style={diamond,draw=black!70,fill=white,aspect=1.8,
    align=center,text width=53pt,inner sep=1.5pt},

]

\node[stbox,text width=58pt] (off) at (100,0) {T1\\thinking off\\Fixed gate};

\node[stbox,text width=76pt,fill=blue!5,draw=blue!55!black] (on) at (190,-77)

\node[stbox,text width=67pt] (final) at (296,-77) {Final link\\or abstention};

\draw[stflow] (input.east) -- (off.west);
\draw[stflow] (off.east) -- (null.west);
\draw[stflow] (null.east) -- node[above,inner sep=2pt] {No} (keep.west);

\draw[stflow] (keep.south) -- (final.north);
\draw[stflow] (on.east) -- (final.west);
